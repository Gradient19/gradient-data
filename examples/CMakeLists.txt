cmake_minimum_required(VERSION 3.16)
project(gradient_data_customer_examples LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
find_package(GradientData CONFIG REQUIRED)
add_executable(read_range_c read_range.c)
target_link_libraries(read_range_c PRIVATE GradientData::shared)
add_executable(read_range_cpp read_range.cpp)
target_link_libraries(read_range_cpp PRIVATE GradientData::static)
# The Windows static SDK was built with the static MSVC runtime.
if(MSVC)
  set_property(TARGET read_range_cpp PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
endif()

add_executable(read_member_c read_member.c)
target_link_libraries(read_member_c PRIVATE GradientData::shared)
add_executable(read_member_cpp read_member.cpp)
target_link_libraries(read_member_cpp PRIVATE GradientData::static)
if(MSVC)
  set_property(TARGET read_member_cpp PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
endif()

# Compile the same bounded C ABI consumer as shared C11 and static C++17.
add_executable(runtime_asset_reader_c runtime_asset_reader.c)
target_link_libraries(runtime_asset_reader_c PRIVATE GradientData::shared)
configure_file(runtime_asset_reader.c runtime_asset_reader.cpp COPYONLY)
add_executable(runtime_asset_reader_cpp "${CMAKE_CURRENT_BINARY_DIR}/runtime_asset_reader.cpp")
target_link_libraries(runtime_asset_reader_cpp PRIVATE GradientData::static)
if(MSVC)
  set_property(TARGET runtime_asset_reader_cpp PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
endif()
